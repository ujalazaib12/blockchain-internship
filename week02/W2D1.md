pragma solidity ^0.8.0; — tells the compiler which Solidity version to use
contract SimpleStorage { } — defines your contract, everything inside the curly braces belongs to it
uint public myNumber; — a state variable to store a number, public means anyone can read it
address public owner; — stores a wallet address, we'll use it to record who deployed the contract
bool public isActive; — a true/false flag
string public myText; — stores text
constructor() { } — special function that runs only once, when the contract is deployed
msg.sender — a built-in variable that means "whoever is calling this function right now"
setNumber — a function to change myNumber
getNumber — a function to read myNumber, marked view because it only reads data, doesn't change anything 

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.0;

contract SimpleStorage {
    uint public myNumber;
    address public owner;
    bool public isActive;
    string public myText;

    constructor() {
        owner = msg.sender;
        isActive = true;
        myText = "Hello Solidity";
    }

    function setNumber(uint _num) public {
        myNumber = _num;
    }

    function getNumber() public view returns (uint) {
        return myNumber;
    }
}
